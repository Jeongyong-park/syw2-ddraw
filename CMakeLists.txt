cmake_minimum_required(VERSION 3.24)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" HQCDD_VERSION)
string(STRIP "${HQCDD_VERSION}" HQCDD_VERSION)
if(NOT HQCDD_VERSION MATCHES "^(0|[1-9][0-9]*)[.](0|[1-9][0-9]*)[.](0|[1-9][0-9]*)$")
  message(FATAL_ERROR "VERSION must be a normal SemVer MAJOR.MINOR.PATCH without leading zeroes")
endif()
foreach(part "${CMAKE_MATCH_1}" "${CMAKE_MATCH_2}" "${CMAKE_MATCH_3}")
  if(part GREATER 65535)
    message(FATAL_ERROR "VERSION components must fit Windows 16-bit version fields")
  endif()
endforeach()
project(syw2_ddraw VERSION ${HQCDD_VERSION} LANGUAGES CXX RC)
configure_file(src/version.h.in generated/version.h @ONLY)
configure_file(src/version.rc.in generated/version.rc @ONLY)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "SYW2Plus requires Windows x86: cmake -A Win32")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(HQCDD_SOURCES src/ddraw.cpp src/display_settings.cpp src/gdi_child.cpp src/pixels.cpp src/gpu.cpp src/overlay.cpp src/overlay_input.cpp src/osd.cpp src/widescreen_runtime.cpp src/settings.rc "${CMAKE_CURRENT_BINARY_DIR}/generated/version.rc")
add_library(hqcdd SHARED ${HQCDD_SOURCES} exports.def)
add_library(hqcdd_asi SHARED ${HQCDD_SOURCES} exports-asi.def)
set_target_properties(hqcdd_asi PROPERTIES OUTPUT_NAME hqcdd SUFFIX .asi ARCHIVE_OUTPUT_NAME hqcdd_asi PDB_NAME hqcdd_asi)
target_compile_definitions(hqcdd_asi PRIVATE HQCDD_ASI)
foreach(target hqcdd hqcdd_asi)
  target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX DIRECTDRAW_VERSION=0x0700)
  target_link_libraries(${target} PRIVATE psapi gdi32 user32 comctl32 dxguid ole32 d3d11 dxgi d3dcompiler bcrypt)
  target_compile_options(${target} PRIVATE /W4 /utf-8 /EHsc)
endforeach()
add_executable(wrapper_test tests/wrapper_test.cpp src/pixels.cpp)
target_compile_definitions(wrapper_test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX DIRECTDRAW_VERSION=0x0700)
target_compile_options(wrapper_test PRIVATE /W4 /utf-8 /EHsc)
target_link_libraries(wrapper_test PRIVATE user32 gdi32 dxguid ole32)
enable_testing()

add_test(NAME asi_core_test COMMAND wrapper_test $<TARGET_FILE:hqcdd_asi>)
