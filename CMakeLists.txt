cmake_minimum_required(VERSION 3.24)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" HQCDD_VERSION)
string(STRIP "${HQCDD_VERSION}" HQCDD_VERSION)
if(NOT HQCDD_VERSION MATCHES "^(0|[1-9][0-9]*)[.](0|[1-9][0-9]*)[.](0|[1-9][0-9]*)$")
  message(FATAL_ERROR "VERSION must be a normal SemVer MAJOR.MINOR.PATCH without leading zeroes")
endif()
foreach(part "${CMAKE_MATCH_1}" "${CMAKE_MATCH_2}" "${CMAKE_MATCH_3}")
  if(part GREATER 65535)
    message(FATAL_ERROR "VERSION components must fit Windows 16-bit version fields")
  endif()
endforeach()
project(syw2_ddraw VERSION ${HQCDD_VERSION} LANGUAGES CXX RC)
configure_file(src/version.h.in generated/version.h @ONLY)
configure_file(src/version.rc.in generated/version.rc @ONLY)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "SYW2Plus requires Windows x86: cmake -A Win32")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(hqcdd SHARED src/ddraw.cpp src/pixels.cpp src/gpu.cpp src/overlay.cpp src/osd.cpp src/settings.rc exports.def "${CMAKE_CURRENT_BINARY_DIR}/generated/version.rc")
target_include_directories(hqcdd PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_definitions(hqcdd PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX DIRECTDRAW_VERSION=0x0700)
target_link_libraries(hqcdd PRIVATE psapi gdi32 user32 comctl32 dxguid ole32 d3d11 dxgi d3dcompiler)
target_compile_options(hqcdd PRIVATE /W4 /utf-8 /EHsc)
add_executable(wrapper_test tests/wrapper_test.cpp src/pixels.cpp)
target_compile_definitions(wrapper_test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX DIRECTDRAW_VERSION=0x0700)
target_compile_options(wrapper_test PRIVATE /W4 /utf-8 /EHsc)
target_link_libraries(wrapper_test PRIVATE user32 gdi32 dxguid ole32)
enable_testing()
add_test(NAME wrapper_test COMMAND wrapper_test $<TARGET_FILE:hqcdd>)
add_executable(gpu_test tests/gpu_test.cpp src/gpu.cpp src/pixels.cpp)
target_compile_definitions(gpu_test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(gpu_test PRIVATE /W4 /utf-8 /EHsc)
target_link_libraries(gpu_test PRIVATE user32 d3d11 dxgi d3dcompiler)
add_test(NAME gpu_test COMMAND gpu_test)

# Smoke test links the system import library; the packaged DLL is placed beside it.
add_executable(loader_test tests/loader_test.cpp)
target_compile_definitions(loader_test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(loader_test PRIVATE /W4 /utf-8 /EHsc)
target_link_libraries(loader_test PRIVATE ddraw dxguid user32 ole32)

# CPU-only probe for Unicorn. Never included in user distribution.
add_library(unicorn_probe SHARED tests/unicorn_probe.cpp)
target_compile_options(unicorn_probe PRIVATE /W4 /utf-8 /EHsc)

add_executable(osd_stats_test tests/osd_stats_test.cpp)
add_test(NAME osd_stats_test COMMAND osd_stats_test)
