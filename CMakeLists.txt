cmake_minimum_required(VERSION 3.24)
project(syw2_ddraw VERSION 0.5.2 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "SYW2Plus requires Windows x86: cmake -A Win32")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(hqcdd SHARED src/ddraw.cpp src/pixels.cpp src/gpu.cpp src/overlay.cpp src/settings.rc exports.def)
target_compile_definitions(hqcdd PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX DIRECTDRAW_VERSION=0x0700)
target_link_libraries(hqcdd PRIVATE gdi32 user32 comctl32 dxguid ole32 d3d11 dxgi d3dcompiler)
target_compile_options(hqcdd PRIVATE /W4 /utf-8 /EHsc)
add_executable(wrapper_test tests/wrapper_test.cpp src/pixels.cpp)
target_compile_definitions(wrapper_test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX DIRECTDRAW_VERSION=0x0700)
target_compile_options(wrapper_test PRIVATE /W4 /utf-8 /EHsc)
target_link_libraries(wrapper_test PRIVATE user32 gdi32 dxguid ole32)
enable_testing()
add_test(NAME wrapper_test COMMAND wrapper_test $<TARGET_FILE:hqcdd>)
add_executable(gpu_test tests/gpu_test.cpp src/gpu.cpp src/pixels.cpp)
target_compile_definitions(gpu_test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(gpu_test PRIVATE /W4 /utf-8 /EHsc)
target_link_libraries(gpu_test PRIVATE user32 d3d11 dxgi d3dcompiler)
add_test(NAME gpu_test COMMAND gpu_test)
