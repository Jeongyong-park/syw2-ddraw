# Keep build policy local to each target; do not propagate it to consumers.
function(hqcdd_compile_options target)
  target_compile_options(${target} PRIVATE /W4 /utf-8 /EHsc)
endfunction()

function(hqcdd_windows_definitions target)
  target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX ${ARGN})
endfunction()

# Every native test chooses its execution environment when it is registered.
function(hqcdd_test name environment)
  if(NOT environment MATCHES "^(ci|desktop)$")
    message(FATAL_ERROR "Unknown test environment: ${environment}")
  endif()
  add_test(NAME ${name} COMMAND ${ARGN})
  set_tests_properties(${name} PROPERTIES LABELS "native;${environment}")
endfunction()
